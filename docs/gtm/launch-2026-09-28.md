### Queries (read-only)

Supabase SQL editor. They only read. Checked for syntax and logic on a local Postgres with the
same columns, not run against production. Exclude your own test accounts by email.

```sql
-- 1. Sign-ups, installs and activations by sign-up day (ET), since launch.
--    Activation = the first published edit on a site the account owns: the
--    same signal as "An edit published" on the dashboard checklist.
with signups as (
  select id as user_id, email_confirmed_at as at
  from auth.users
  where email_confirmed_at >= timestamptz '2026-09-28 00:00 America/New_York'
),
owned as (
  select sp.user_id, sp.site_id
  from public.site_permissions sp
  join signups s on s.user_id = sp.user_id
  where sp.permission = 'admin'
),
installed as (
  select o.user_id, min(st.live_at) as at
  from owned o
  join public.sites st on st.id = o.site_id
  group by o.user_id
),
activated as (
  select o.user_id, min(h.created_at) as at
  from owned o
  join public.content_elements ce on ce.site_id = o.site_id
  join public.staging_history h on h.content_element_id = ce.id
  where h.action = 'publish'
  group by o.user_id
)
select (s.at at time zone 'America/New_York')::date as signup_day_et,
  count(*)    as signups,
  count(i.at) as installed,
  count(a.at) as activated,
  round((percentile_cont(0.5) within group (
    order by extract(epoch from a.at - s.at) / 60))::numeric, 1)
    as p50_minutes_to_first_publish
from signups s
left join installed i on i.user_id = s.user_id
left join activated a on a.user_id = s.user_id
group by 1
order by 1;

-- 2. Spots cross-check: the first 20 accounts to complete sign-in after s47
--    went live. Set the timestamp to s47's go-live. The homepage counter is
--    what visitors see; if the two disagree, quote the counter and log the gap.
select row_number() over (order by email_confirmed_at) as spot,
  email, email_confirmed_at
from auth.users
where email_confirmed_at >= timestamptz '2026-09-28 07:00 America/New_York'
order by email_confirmed_at
limit 20;

-- 3. Signed up more than 24 h ago, no site installed yet: email each one.
select u.email, u.email_confirmed_at
from auth.users u
where u.email_confirmed_at >= timestamptz '2026-09-28 00:00 America/New_York'
  and u.email_confirmed_at < now() - interval '24 hours'
  and not exists (
    select 1
    from public.site_permissions sp
    join public.sites st on st.id = sp.site_id
    where sp.user_id = u.id and sp.permission = 'admin' and st.live_at is not null
  )
order by u.email_confirmed_at;
```
